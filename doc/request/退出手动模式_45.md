/** 
 * 根据core返回结果判断本次退出手动模式是否成功
 */
fun QuitManualMode.getQuitResult(): Pair = when {
    !isManualMode -> true to ""
    isUnrecoverableError -> false to "程序错误，退出手动模式失败"
    isUrgencyStop -> false to "急停状态，退出手动模式失败"
    isCharging -> true to ""
    !isWorkMode -> false to "为了不影响后续任务执行，请将机器人推回基站"
    isOffMap || isRestrictedZone || !isMaxPassable || !isPlanPath -> false to "请将机器人推到开阔的清洁区域后，再次尝试"
    else -> true to ""
}
